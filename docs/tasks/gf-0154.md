(gf#40), следующий шаг из его «Чем возобновлять» — извлечение в наш формат.

- Папки листов для таблицы (скрипт, 18.10):

```python
"""Папки листов для таблицы gf#154: эталон BuildingNet, генератор синтетики, подмены на BuildingNet."""
import sys
from pathlib import Path
import numpy
from genfacade import config
from genfacade.metrics import calibrate
from genfacade.schema import Mode, Source, Split
from genfacade.train.tokens import given_sides
cfg = config.load(); out = Path(sys.argv[1])
real = calibrate.houses(Source.BUILDINGNET, Split.TEST, 0, cfg)
synth = calibrate.houses(Source.SYNTHETIC, Split.VAL, 0, cfg)
for mode in Mode:
    sheets = [given_sides(s, mode) for s in real.values()]
    calibrate.write(out / mode / "reference", list(real), sheets)
    calibrate.write(out / mode / "generator", list(synth), [given_sides(s, mode) for s in synth.values()])
    for name in ("random", "foreign", "empty"):
        calibrate.write(out / mode / name, list(real), calibrate.ROWS[name](sheets, numpy.random.default_rng(cfg.metrics.seed)))
print(len(real), len(synth))
```
